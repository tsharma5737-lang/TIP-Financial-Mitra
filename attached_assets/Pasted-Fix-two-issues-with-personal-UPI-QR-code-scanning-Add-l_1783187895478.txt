javascript// Personal UPI indicators
const PERSONAL_UPI_PATTERNS = [
  /@okaxis$/, /@okhdfcbank$/, /@okicici$/, /@oksbi$/,
  /@ybl$/, /@ibl$/, /@axl$/, /@rapl$/,
  /@paytm$/, /@apl$/, /^\d{10}@/
];

// Merchant UPI indicators  
const MERCHANT_UPI_PATTERNS = [
  /^[a-z]+\.rzp@/, /^[a-z]+\.paytm@/,
  /merchant/i, /store/i, /shop/i, /mart/i
];

function isPersonalUPI(vpa) {
  if (!vpa) return false;
  // 10-digit mobile number VPAs are always personal
  if (/^\d{10}@/.test(vpa)) return true;
  // Check merchant indicators first
  for (const pattern of MERCHANT_UPI_PATTERNS) {
    if (pattern.test(vpa)) return false;
  }
  // Check personal indicators
  for (const pattern of PERSONAL_UPI_PATTERNS) {
    if (pattern.test(vpa)) return true;
  }
  return false;
}
